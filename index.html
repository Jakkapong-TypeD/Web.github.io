import machine import time import network import socket # 1. ตั้งค่า Wi-Fi Hotspot (เช็กว่าปล่อย Hotspot จากโทรศัพท์ไว้แล้ว) WIFI_SSID = "English_or_Spanish" WIFI_PASS = "123456789" # 2. ชื่อไอดีสำหรับส่งเข้าเว็บ GitHub (ตั้งชื่อว่า jakkapong-mq2-data เพื่อให้ตรงกับหน้าเว็บ) DEVICE_ID = "jakkapong-mq2-data" # 3. ตั้งค่าเซนเซอร์ MQ-2 (พอร์ต I1 ของ KidBright32iP คือขา GPIO36) mq2_pin = machine.ADC(machine.Pin(36)) mq2_pin.atten(machine.ADC.ATTN_11DB) # 4. เริ่มต้นเชื่อมต่อ Wi-Fi wlan = network.WLAN(network.STA_IF) wlan.active(True) wlan.connect(WIFI_SSID, WIFI_PASS) while not wlan.isconnected(): time.sleep_ms(500) print("Wi-Fi Connected!") while True: gas_value = mq2_pin.read() # ใช้โครงสร้าง Socket พื้นฐานยิงข้อมูลเข้าอินเทอร์เน็ตตรงๆ try: ai = socket.getaddrinfo("dweet.io", 80) addr = ai[-1] s = socket.socket() s.connect(addr) path = "/dweet/for/" + DEVICE_ID + "?gas=" + str(gas_value) req = "GET " + path + " HTTP/1.1\r\nHost: dweet.io\r\nConnection: close\r\n\r\n" s.send(bytes(req, 'utf8')) s.close() print("ส่งข้อมูลเข้า Cloud สำเร็จ ค่าปัจจุบัน:", gas_value) except: print("เครือข่ายขัดข้อง ส่งข้อมูลไม่สำเร็จ") time.sleep(2) # วนลูปส่งค่าทุกๆ 2 วินาที
